CREATE DATABASE PlayTogether;
USE PlayTogether;

CREATE TABLE Users (
    UserID INT PRIMARY KEY AUTO_INCREMENT,
    Name VARCHAR(100) NOT NULL,
    Phone VARCHAR(15),
    Email VARCHAR(100) UNIQUE
);
CREATE TABLE SportsGrounds (
    GroundID INT PRIMARY KEY AUTO_INCREMENT,
    GroundName VARCHAR(100) NOT NULL,
    SportType VARCHAR(50) NOT NULL,
    Location VARCHAR(100) NOT NULL,
    HourlyRate DECIMAL(10,2) NOT NULL,
    Capacity INT DEFAULT 10,
    Status ENUM('Active','Inactive') DEFAULT 'Active',
    CHECK (HourlyRate > 0)
);
CREATE TABLE Slots (
    SlotID INT PRIMARY KEY AUTO_INCREMENT,
    GroundID INT NOT NULL,
    SlotDate DATE NOT NULL,
    StartTime TIME NOT NULL,
    EndTime TIME NOT NULL,
    SlotStatus ENUM('Available','Booked') DEFAULT 'Available',
    FOREIGN KEY (GroundID) REFERENCES SportsGrounds(GroundID) ON DELETE CASCADE,
    CHECK (EndTime > StartTime),
    UNIQUE (GroundID, SlotDate, StartTime, EndTime)
);
CREATE TABLE Bookings (
    BookingID INT PRIMARY KEY AUTO_INCREMENT,
    UserID INT NOT NULL,
    SlotID INT NOT NULL,
    BookingDate DATE NOT NULL,
    BookingStatus ENUM('Pending','Confirmed','Cancelled','Completed') DEFAULT 'Pending',
    TotalAmount DECIMAL(10,2) NOT NULL,
    FOREIGN KEY (UserID) REFERENCES Users(UserID),
    FOREIGN KEY (SlotID) REFERENCES Slots(SlotID),
    CHECK (TotalAmount > 0)
);
CREATE TABLE Payments (
    PaymentID INT PRIMARY KEY AUTO_INCREMENT,
    BookingID INT NOT NULL,
    Amount DECIMAL(10,2) NOT NULL,
    PaymentMethod ENUM('UPI','Card','NetBanking','Cash'),
    PaymentStatus ENUM('Pending','Success','Failed') DEFAULT 'Pending',
    PaymentDate DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (BookingID) REFERENCES Bookings(BookingID),
    CHECK (Amount > 0)
);
CREATE TABLE Payment_Audit (
    AuditID INT PRIMARY KEY AUTO_INCREMENT,
    PaymentID INT,
    ActionType VARCHAR(20),
    OldStatus VARCHAR(20),
    NewStatus VARCHAR(20),
    Amount DECIMAL(10,2),
    ActionTime DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- PART 5 - VIEWS

CREATE OR REPLACE VIEW view_ground_overview AS
SELECT GroundID, GroundName, SportType, Location, HourlyRate, Status FROM SportsGrounds;

CREATE OR REPLACE VIEW view_booking_details AS
SELECT b.BookingID, u.Name AS UserName, g.GroundName, g.SportType, s.SlotDate, s.StartTime, s.EndTime, b.BookingStatus, b.TotalAmount
FROM Bookings b JOIN Users u ON b.UserID = u.UserID JOIN Slots s ON b.SlotID = s.SlotID JOIN SportsGrounds g ON s.GroundID = g.GroundID;

CREATE OR REPLACE VIEW view_payment_summary AS
SELECT p.PaymentID, p.BookingID, u.Name AS UserName, p.Amount, p.PaymentMethod, p.PaymentStatus, p.PaymentDate
FROM Payments p JOIN Bookings b ON p.BookingID = b.BookingID JOIN Users u ON b.UserID = u.UserID;

CREATE OR REPLACE VIEW view_ground_booking_summary AS
SELECT g.GroundID, g.GroundName, COUNT(b.BookingID) AS TotalBookings, COALESCE(SUM(CASE WHEN b.BookingStatus = 'Confirmed' THEN b.TotalAmount ELSE 0 END),0) AS ConfirmedBookingValue
FROM SportsGrounds g LEFT JOIN Slots s ON g.GroundID = s.GroundID LEFT JOIN Bookings b ON s.SlotID = b.SlotID GROUP BY g.GroundID, g.GroundName;

-- PART 7 - TRIGGERS
-- ============================================================
DELIMITER //
CREATE TRIGGER trg_payment_amount_check BEFORE INSERT ON Payments FOR EACH ROW BEGIN IF NEW.Amount <= 0 THEN SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Payment amount must be greater than zero'; END IF; END //
CREATE TRIGGER trg_payment_update_audit AFTER UPDATE ON Payments FOR EACH ROW BEGIN IF OLD.PaymentStatus <> NEW.PaymentStatus THEN INSERT INTO Payment_Audit (PaymentID, ActionType, OldStatus, NewStatus, Amount) VALUES (NEW.PaymentID, 'UPDATE', OLD.PaymentStatus, NEW.PaymentStatus, NEW.Amount); END IF; END //
CREATE TRIGGER trg_payment_delete_audit AFTER DELETE ON Payments FOR EACH ROW BEGIN INSERT INTO Payment_Audit (PaymentID, ActionType, OldStatus, NewStatus, Amount) VALUES (OLD.PaymentID, 'DELETE', OLD.PaymentStatus, NULL, OLD.Amount); END //
CREATE TRIGGER trg_booking_cancel_slot AFTER UPDATE ON Bookings FOR EACH ROW BEGIN IF NEW.BookingStatus = 'Cancelled' AND OLD.BookingStatus <> 'Cancelled' THEN UPDATE Slots SET SlotStatus = 'Available' WHERE SlotID = NEW.SlotID; UPDATE Payments SET PaymentStatus = 'Failed' WHERE BookingID = NEW.BookingID AND PaymentStatus = 'Pending'; END IF; END //
DELIMITER ;
